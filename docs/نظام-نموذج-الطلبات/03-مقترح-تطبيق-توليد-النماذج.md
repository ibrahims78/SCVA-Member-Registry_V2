### الخطوة 1.3 — Zustand Store

```typescript
// src/store/projectStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { ProjectConfig, FormField } from '../types';

interface ProjectStore {
  config     : ProjectConfig | null;
  isDirty    : boolean;

  // Actions
  initProject   : (partial?: Partial<ProjectConfig>) => void;
  updateConfig  : (updates: Partial<ProjectConfig>) => void;
  addField      : (field: FormField) => void;
  updateField   : (id: string, updates: Partial<FormField>) => void;
  removeField   : (id: string) => void;
  reorderFields : (from: number, to: number) => void;
  resetProject  : () => void;
  loadFromFile  : (json: string) => void;
}

export const useProjectStore = create<ProjectStore>()(
  persist(
    (set, get) => ({
      config  : null,
      isDirty : false,

      initProject: (partial = {}) => set({
        config: {
          orgName      : '',
          adminEmail   : '',
          spreadsheetId: '',
          sheetName    : 'سجل الطلبات',
          timezone     : 'Asia/Damascus',
          emailTemplate: 'corporate',
          accentColor  : '#1565C0',
          logoEmoji    : '🏢',
          fields       : [],
          steps        : [],
          createdAt    : new Date().toISOString(),
          version      : '1.0.0',
          ...partial,
        },
        isDirty: false,
      }),

      updateConfig: (updates) => set((state) => ({
        config : state.config ? { ...state.config, ...updates } : null,
        isDirty: true,
      })),

      addField: (field) => set((state) => ({
        config : state.config
          ? { ...state.config, fields: [...state.config.fields, field] }
          : null,
        isDirty: true,
      })),

      updateField: (id, updates) => set((state) => ({
        config: state.config ? {
          ...state.config,
          fields: state.config.fields.map(f =>
            f.id === id ? { ...f, ...updates } : f
          ),
        } : null,
        isDirty: true,
      })),

      removeField: (id) => set((state) => ({
        config: state.config ? {
          ...state.config,
          fields: state.config.fields.filter(f => f.id !== id),
        } : null,
        isDirty: true,
      })),

      reorderFields: (from, to) => set((state) => {
        if (!state.config) return {};
        const fields = [...state.config.fields];
        const [moved] = fields.splice(from, 1);
        fields.splice(to, 0, moved);
        return { config: { ...state.config, fields }, isDirty: true };
      }),

      resetProject: () => set({ config: null, isDirty: false }),

      loadFromFile: (json) => {
        try {
          const parsed = JSON.parse(json) as ProjectConfig;
          set({ config: parsed, isDirty: false });
        } catch {
          throw new Error('ملف المشروع غير صالح أو تالف');
        }
      },
    }),
    {
      name   : 'form-generator-project',
      version: 1,
    }
  )
);
```

```typescript
// src/lib/excelReader.ts
import * as XLSX from 'xlsx';
import type { FormField } from '../types';
import { guessFieldType, arabicToJsKey } from './fieldGuesser';
import { v4 as uuid } from 'uuid';

export interface ExcelReadResult {
  sheetName   : string;
  totalColumns: number;
  fields      : FormField[];
  warnings    : string[];
}

export async function readExcelFile(file: File): Promise<ExcelReadResult> {
  const buffer   = await file.arrayBuffer();
  const workbook = XLSX.read(buffer, { type: 'array', cellText: true });

  const warnings: string[] = [];
  const sheetName = workbook.SheetNames[0];
  const worksheet = workbook.Sheets[sheetName];

  if (!worksheet['!ref']) {
    throw new Error('الملف فارغ أو لا يحتوي على بيانات');
  }

  const range   = XLSX.utils.decode_range(worksheet['!ref']);
  const fields  : FormField[] = [];
  const usedKeys: Set<string> = new Set();

  // أعمدة تُضاف تلقائياً ولا تؤخذ من Excel
  const AUTO_COLUMNS = [
    'رقم الطلب', 'تاريخ الإرسال', 'وقت الإرسال',
    'الحالة', 'ملاحظات المسؤول', 'تاريخ آخر تحديث',
  ];

  for (let col = range.s.c; col <= range.e.c; col++) {
    const headerCell = worksheet[XLSX.utils.encode_cell({ r: 0, c: col })];
    if (!headerCell?.v) {
      warnings.push(`العمود ${XLSX.utils.encode_col(col)} لا يحتوي على عنوان — تم تجاهله`);
      continue;
    }

    const arabicName = String(headerCell.v).trim();

    // تخطّي الأعمدة التلقائية
    if (AUTO_COLUMNS.some(ac => arabicName.includes(ac.split(' ')[0]))) {
      warnings.push(`عمود "${arabicName}" سيُضاف تلقائياً — لا حاجة لتضمينه`);
      continue;
    }

    // توليد jsKey فريد
    let jsKey = arabicToJsKey(arabicName);
    let keyCounter = 2;
    while (usedKeys.has(jsKey)) jsKey = `${jsKey}${keyCounter++}`;
    usedKeys.add(jsKey);

    // استخراج عينة من البيانات للتخمين
    const sampleValues: string[] = [];
    for (let row = 1; row <= Math.min(5, range.e.r); row++) {
      const cell = worksheet[XLSX.utils.encode_cell({ r: row, c: col })];
      if (cell?.v) sampleValues.push(String(cell.v));
    }

    fields.push({
      id         : uuid(),
      arabicName,
      jsKey,
      excelCol   : XLSX.utils.encode_col(col),
      type       : guessFieldType(arabicName, sampleValues),
      placeholder: '',
      required   : true,
      hidden     : false,
      options    : [],
      validation : ['required'],
      step       : 1,
    });
  }

  return { sheetName, totalColumns: range.e.c - range.s.c + 1, fields, warnings };
}
```

```typescript
// src/lib/fieldGuesser.ts
import type { FieldType } from '../types';

// قاموس الكلمات المفتاحية لكل نوع
const TYPE_PATTERNS: Record<FieldType, RegExp> = {
  tel     : /هاتف|موبايل|جوال|تلفون|phone|mobile|tel/i,
  email   : /بريد|إيميل|ايميل|email/i,
  date    : /تاريخ|يوم|ميلاد|date/i,
  number  : /عمر|رقم|عدد|مبلغ|تكلفة|سعر|number|count|age/i,
  textarea: /ملاحظ|تفاصيل|وصف|شرح|اقتراح|شكوى|مشكلة|طلب|رسالة|note|detail|desc/i,
  select  : /نوع|جنس|فئة|تصنيف|حالة|مستوى|درجة|قسم|type|category|status/i,
  checkbox: /موافق|أوافق|agree|confirm|check/i,
  text    : /.*/, // الافتراضي
};

export function guessFieldType(name: string, samples: string[] = []): FieldType {
  // فحص الاسم أولاً
  for (const [type, pattern] of Object.entries(TYPE_PATTERNS)) {
    if (type !== 'text' && pattern.test(name)) {
      return type as FieldType;
    }
  }

  // فحص عينات البيانات إن وُجدت
  if (samples.length > 0) {
    const allNumeric  = samples.every(s => !isNaN(+s));
    const allDates    = samples.every(s => !isNaN(Date.parse(s)));
    const hasAt       = samples.some(s => s.includes('@'));
    const longTexts   = samples.some(s => s.length > 50);
    const fewUnique   = new Set(samples).size <= 5;

    if (hasAt)       return 'email';
    if (allDates)    return 'date';
    if (allNumeric)  return 'number';
    if (longTexts)   return 'textarea';
    if (fewUnique && samples.length >= 3) return 'select';
  }

  return 'text';
}

// تحويل النص العربي لمفتاح JavaScript صالح
export function arabicToJsKey(arabicName: string): string {
  const MAP: Record<string, string> = {
    'الاسم الكامل': 'fullName', 'الاسم': 'name',
    'رقم الهاتف': 'phone', 'الهاتف': 'phone', 'الجوال': 'mobile',
    'البريد الإلكتروني': 'email', 'البريد': 'email',
    'المدينة': 'city', 'المنطقة': 'region', 'العنوان': 'address',
    'نوع الطلب': 'requestType', 'تفاصيل الطلب': 'details',
    'الملاحظات': 'notes', 'العمر': 'age', 'الجنس': 'gender',
  };

  // فحص الخريطة أولاً
  const trimmed = arabicName.trim();
  if (MAP[trimmed]) return MAP[trimmed];

  // تحويل تلقائي: أزل أل التعريف وحوّل المسافات لـ camelCase
  const words = trimmed
    .replace(/^ال/, '')
    .split(/[\s\/\-\_]+/)
    .map((w, i) => {
      const latin = ARABIC_TO_LATIN[w] || w.charCodeAt(0).toString(16);
      return i === 0 ? latin.toLowerCase() : latin[0]?.toUpperCase() + latin.slice(1);
    });

  return words.join('') || `field_${Math.random().toString(36).slice(2, 6)}`;
}

// خريطة الكلمات الشائعة عربي → إنجليزي
const ARABIC_TO_LATIN: Record<string, string> = {
  'اسم': 'name', 'هاتف': 'phone', 'بريد': 'email', 'مدينة': 'city',
  'عنوان': 'address', 'تاريخ': 'date', 'رقم': 'number', 'نوع': 'type',
  'تفاصيل': 'details', 'ملاحظات': 'notes', 'حالة': 'status',
  'جنس': 'gender', 'عمر': 'age', 'قسم': 'department', 'وصف': 'description',
};
```

```typescript
// src/lib/stepDistributor.ts
import type { FormField, WizardStep } from '../types';

export function distributeIntoSteps(fields: FormField[]): WizardStep[] {
  const visibleFields = fields.filter(f => !f.hidden);

  // المجموعة الشخصية: نص + هاتف + بريد
  const personalTypes = new Set(['text', 'tel', 'email']);
  const personal  = visibleFields.filter(f => personalTypes.has(f.type));
  const details   = visibleFields.filter(f => !personalTypes.has(f.type));

  const steps: WizardStep[] = [];

  if (personal.length > 0) {
    steps.push({
      id    : 'step-personal',
      title : 'بياناتك الشخصية',
      fields: personal.map(f => ({ ...f, step: 1 })),
    });
  }

  // تقسيم الحقول التفصيلية بحد أقصى 4 حقول في الخطوة
  const CHUNK_SIZE = 4;
  for (let i = 0; i < details.length; i += CHUNK_SIZE) {
    const chunk = details.slice(i, i + CHUNK_SIZE);
    steps.push({
      id    : `step-details-${steps.length}`,
      title : steps.length === 1 ? 'تفاصيل طلبك' : `تفاصيل إضافية`,
      fields: chunk.map(f => ({ ...f, step: steps.length + 1 })),
    });
  }

  // الخطوة الأخيرة دائماً: مراجعة وإرسال
  steps.push({
    id      : 'step-review',
    title   : 'مراجعة وإرسال',
    fields  : [],
    isReview: true,
  });

  return steps;
}
```

```typescript
        ${!field.required ? '<span style="color:var(--text-hint);font-weight:400;">(اختياري)</span>' : ''}
      </label>
      <input class="field-input" type="${inputType}" id="${field.jsKey}"
             placeholder="${field.placeholder || ''}"
             autocomplete="${getAutocomplete(field.jsKey)}">
      <div class="field-error" id="err_${field.jsKey}">
        ${getDefaultError(field)}
      </div>
    </div>`;
  }).join('\n');
}

// ديناميكي: دالة validateStep في JavaScript
function buildValidationFunction(steps: WizardStep[]): string {
  const cases = steps
    .filter(s => !s.isReview)
    .map((step, idx) => {
      const fieldChecks = step.fields
        .filter(f => f.required)
        .map(f => {
          const rule = getValidationRule(f);
          return `    valid &= checkRequired('${f.jsKey}', 'err_${f.jsKey}', ${rule});`;
        })
        .join('\n');
      return `  if (step === ${idx + 1}) {\n${fieldChecks}\n  }`;
    })
    .join('\n\n');

  return `function validateStep(step) {\n  let valid = true;\n\n${cases}\n\n  return Boolean(valid);\n}`;
}

// ديناميكي: جمع البيانات قبل الإرسال
function buildPayloadBuilder(fields: FormField[]): string {
  const visibleFields = fields.filter(f => !f.hidden);
  const lines = visibleFields
    .map(f => `    ${f.jsKey}: document.getElementById('${f.jsKey}').value.trim(),`)
    .join('\n');
  return `const payload = {\n${lines}\n  };`;
}
```

```typescript
  const [device, setDevice] = useState<'desktop' | 'mobile'>('desktop');

    if (blobUrlRef.current) URL.revokeObjectURL(blobUrlRef.current);

    return () => {
```

### الخطوة 7.1 — Autosave وحماية البيانات

```typescript
// في App.tsx — تحذير عند محاولة إغلاق الصفحة مع تغييرات غير محفوظة
useEffect(() => {
  const handler = (e: BeforeUnloadEvent) => {
    if (isDirty) {
      e.preventDefault();
      e.returnValue = '';
    }
  };
  window.addEventListener('beforeunload', handler);
  return () => window.removeEventListener('beforeunload', handler);
}, [isDirty]);

// Autosave كل 60 ثانية (Zustand persist يتولى ذلك تلقائياً)
```

### الخطوة 7.3 — تحقق الأمان

```typescript
// تنظيف المُدخَلات قبل حقنها في القوالب — منع XSS في الكود المُولَّد
function sanitizeForCodeInjection(str: string): string {
  return str
    .replace(/`/g, "'")           // backticks تكسر Template Literals
    .replace(/\\/g, '\\\\')       // backslash
    .replace(/\$\{/g, '\\${')     // منع حقن template expressions
    .replace(/</g, '&lt;')        // منع HTML
    .replace(/>/g, '&gt;')
    .trim()
    .slice(0, 200);               // حد أقصى للطول
}

// التطبيق: كل قيمة من المستخدم تمر عبر هذه الدالة قبل الحقن في القوالب
```
