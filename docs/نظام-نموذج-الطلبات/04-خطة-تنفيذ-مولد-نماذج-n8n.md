## لماذا n8n بدلاً من Apps Script؟

| المعيار | Apps Script (ملف 03) | n8n (هذا الملف) |
|---------|---------------------|----------------|
| **تعديل الـ Workflow** | تعديل كود + نشر إصدار جديد | سحب وإفلات بدون كود |
| **حصة البريد** | 100 إيميل/يوم | لا حدود |
| **تتبع الأخطاء** | logs نصية فقط | واجهة مرئية لكل execution |
| **إضافة قناة جديدة** | كود جديد معقد | node جاهز (Slack/واتساب/SMS) |
| **إعادة المحاولة** | يدوي | تلقائي عند الفشل |
| **CORS** | يستلزم mode:no-cors | مدعوم بالكامل (قراءة الرد ممكنة) |
| **مصادر البيانات** | Google فقط | 400+ خدمة |

### الخطوة 1.3 — Zustand Store المُكيَّف لـ n8n

```typescript
// src/store/projectStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { ProjectConfig, FormField, N8nConfig } from '../types/project';

interface ProjectStore {
  config  : ProjectConfig | null;
  isDirty : boolean;

  initProject      : (partial?: Partial<ProjectConfig>) => void;
  updateConfig     : (updates: Partial<ProjectConfig>) => void;
  updateN8nConfig  : (updates: Partial<N8nConfig>) => void;
  addField         : (field: FormField) => void;
  updateField      : (id: string, updates: Partial<FormField>) => void;
  removeField      : (id: string) => void;
  reorderFields    : (from: number, to: number) => void;
  addExtraChannel  : (channel: ExtraChannel) => void;
  removeExtraChannel: (type: string) => void;
  resetProject     : () => void;
  loadFromFile     : (json: string) => void;
}

const DEFAULT_N8N_CONFIG: N8nConfig = {
  webhookPath     : 'form-requests',
  spreadsheetId   : '',
  sheetName       : 'سجل الطلبات',
  adminEmail      : '',
  gmailCredential : 'Gmail Account',
  sheetsCredential: 'Google Sheets Account',
  timezone        : 'Asia/Damascus',
  emailTemplate   : 'corporate',
  enableUserEmail : true,
  extraChannels   : [],
};

export const useProjectStore = create<ProjectStore>()(
  persist(
    (set) => ({
      config  : null,
      isDirty : false,

      initProject: (partial = {}) => set({
        config: {
          orgName    : '',
          accentColor: '#1565C0',
          logoEmoji  : '🏢',
          fields     : [],
          steps      : [],
          n8n        : { ...DEFAULT_N8N_CONFIG },
          createdAt  : new Date().toISOString(),
          version    : '1.0.0',
          ...partial,
        },
        isDirty: false,
      }),

      updateN8nConfig: (updates) => set((state) => ({
        config: state.config ? {
          ...state.config,
          n8n: { ...state.config.n8n, ...updates },
        } : null,
        isDirty: true,
      })),

      // ... بقية الـ actions مماثلة لملف 03
    }),
    { name: 'form-gen-n8n-project', version: 1 }
  )
);
```

```typescript
// src/lib/excelReader.ts
import * as XLSX from 'xlsx';
import { guessFieldType, arabicToJsKey } from './fieldGuesser';
import { v4 as uuid } from 'uuid';
import type { FormField } from '../types/project';

export interface ExcelReadResult {
  sheetName   : string;
  totalColumns: number;
  fields      : FormField[];
  warnings    : string[];
  suggestedSheetName: string;  // ← للـ n8n config
}

// الأعمدة التي تُضاف تلقائياً في n8n Workflow ولا تؤخذ من Excel
const AUTO_COLUMNS = [
  'رقم الطلب', 'تاريخ الإرسال', 'وقت الإرسال',
  'الحالة', 'ملاحظات المسؤول', 'تاريخ آخر تحديث',
];

export async function readExcelFile(file: File): Promise<ExcelReadResult> {
  const buffer   = await file.arrayBuffer();
  const workbook = XLSX.read(buffer, { type: 'array', cellText: true });

  const warnings  : string[] = [];
  const sheetName  = workbook.SheetNames[0];
  const worksheet  = workbook.Sheets[sheetName];

  if (!worksheet['!ref']) throw new Error('الملف فارغ أو لا يحتوي على بيانات');

  const range   = XLSX.utils.decode_range(worksheet['!ref']);
  const fields  : FormField[] = [];
  const usedKeys: Set<string> = new Set();

  for (let col = range.s.c; col <= range.e.c; col++) {
    const cell = worksheet[XLSX.utils.encode_cell({ r: 0, c: col })];
    if (!cell?.v) {
      warnings.push(`العمود ${XLSX.utils.encode_col(col)}: لا يوجد عنوان — تم تجاهله`);
      continue;
    }

    const arabicName = String(cell.v).trim();

    if (AUTO_COLUMNS.some(ac => arabicName.includes(ac.split(' ')[0]))) {
      warnings.push(`"${arabicName}": سيُضاف تلقائياً بواسطة n8n — لا حاجة لتضمينه`);
      continue;
    }

    let jsKey    = arabicToJsKey(arabicName);
    let counter  = 2;
    while (usedKeys.has(jsKey)) jsKey = `${jsKey}${counter++}`;
    usedKeys.add(jsKey);

    const samples: string[] = [];
    for (let row = 1; row <= Math.min(5, range.e.r); row++) {
      const sc = worksheet[XLSX.utils.encode_cell({ r: row, c: col })];
      if (sc?.v) samples.push(String(sc.v));
    }

    fields.push({
      id         : uuid(),
      arabicName,
      jsKey,
      excelCol   : XLSX.utils.encode_col(col),
      type       : guessFieldType(arabicName, samples),
      placeholder: '',
      required   : true,
      hidden     : false,
      options    : [],
      validation : ['required'],
      step       : 1,
    });
  }

  return {
    sheetName,
    totalColumns      : range.e.c - range.s.c + 1,
    fields,
    warnings,
    suggestedSheetName: `سجل الطلبات — ${sheetName}`,
  };
}
```

```typescript
${requiredFields}
};

              + String(now.getMonth() + 1).padStart(2, '0');
  const seq   = String(Math.floor(Math.random() * 9000) + 1000);

}

function getLocalTime(timezone) {
  const now     = new Date();

  });
  const timeStr = now.toLocaleTimeString('ar-EG', {

  return { dateStr, timeStr };
}

const { dateStr, timeStr } = getLocalTime('${timezone}');
const requestNumber = generateRequestNumber();

return [{
  json: {
    ...body,
    requestNumber,
    submissionDate: dateStr,

`;
```

```typescript
// src/lib/generators/n8nNodes/sheetsNode.ts
import type { N8nConfig, FormField } from '../../../types/project';

export function buildSheetsNode(
  n8n     : N8nConfig,
  fields  : FormField[]
): N8nNode {
  // بناء خريطة الأعمدة ديناميكياً
  // A=رقم الطلب, B=التاريخ, C=الوقت, D→...=حقول المستخدم, آخر=الحالة
  const FIXED_PREFIX = [
    { column: 'A', fieldKey: 'requestNumber' },
    { column: 'B', fieldKey: 'submissionDate' },
    { column: 'C', fieldKey: 'submissionTime' },
  ];

  const userColumns = fields.map((f, i) => ({
    column  : String.fromCharCode(68 + i), // D, E, F...
    fieldKey: f.jsKey,
  }));

  const lastColIdx = 68 + fields.length;
  const FIXED_SUFFIX = [
    { column: String.fromCharCode(lastColIdx),     fieldKey: 'status' },
    { column: String.fromCharCode(lastColIdx + 1), fieldKey: '' },  // ملاحظات
  ];

  const allColumns = [...FIXED_PREFIX, ...userColumns, ...FIXED_SUFFIX];

  // بناء columnsUi لـ n8n
  const columnsUi = {
    mappingMode: 'defineBelow',
    value: allColumns
      .filter(c => c.fieldKey)
      .map(c => ({
        column : c.column,
        value  : `={{ $json.${c.fieldKey} }}`,
      })),
  };

  return {
    id         : uuid(),
    name       : 'Save to Google Sheets',
    type       : 'n8n-nodes-base.googleSheets',
    typeVersion: 4,
    position   : [680, 300],
    parameters : {
      operation  : 'appendOrUpdate',
      documentId : { value: n8n.spreadsheetId, mode: 'id' },
      sheetName  : { value: n8n.sheetName, mode: 'name' },
      columns    : columnsUi,
      options    : {},
    },
    credentials: {
      googleSheetsOAuth2Api: {
        id  : 'sheets-cred-id',
        name: n8n.sheetsCredential,
      },
    },
  };
}
```

### الخطوة 5.1 — الفارق الجوهري في form.html عن نسخة Apps Script

```
نسخة Apps Script:
fetch(GAS_URL, { method: 'POST', mode: 'no-cors', body: ... })
← لا يمكن قراءة الرد (no-cors)
← لا نعرف هل نجح أم لا؟
← رقم الطلب يُولَّد في العميل (غير موثوق)

نسخة n8n:
fetch(N8N_WEBHOOK_URL, { method: 'POST', body: ... })  // بدون mode:no-cors
← نقرأ الرد الكامل
← { status, message, requestNumber } من الـ Backend
← رقم الطلب موثوق 100% من الـ Backend
```

### الخطوة 5.2 — منطق الإرسال في form.html

```javascript
// الكود المُولَّد داخل form.html

async function submitForm(payload) {
  const submitBtn = document.getElementById('submitBtn');
  submitBtn.disabled = true;
  submitBtn.textContent = '⏳ جارٍ إرسال طلبك...';

  try {
    const response = await fetch('__N8N_WEBHOOK_URL__', {
      method : 'POST',
      headers: { 'Content-Type': 'application/json' },
      body   : JSON.stringify(payload),
    });

    // نقرأ الرد — ممكن لأن n8n يُرسل CORS headers
    const result = await response.json();

    if (result.status === 'success') {
      showSuccessPage(result.requestNumber);
    } else {
      showError(result.message || 'حدث خطأ غير متوقع');
      submitBtn.disabled = false;
      submitBtn.textContent = 'إرسال الطلب';
    }

  } catch (err) {
    showError('تعذّر الاتصال بالخادم. تحقق من اتصالك بالإنترنت.');
    submitBtn.disabled = false;
    submitBtn.textContent = 'إرسال الطلب';
  }
}

// صفحة النجاح مع رقم الطلب الحقيقي من الـ Backend

  document.getElementById('formWrapper').style.display = 'none';
  document.getElementById('successPage').style.display = 'block';

}
```

```typescript
// src/lib/sanitizers.ts

// منع حقن كود في الـ Code Node داخل n8n
export function sanitizeForCodeInjection(str: string): string {
  return str
    .replace(/`/g,    "'")          // يكسر Template Literals
    .replace(/\\/g,   '\\\\')       // backslash
    .replace(/\$\{/g, '\\${')       // Template expressions
    .replace(/</g,    '&lt;')
    .replace(/>/g,    '&gt;')
    .replace(/"/g,    '\\"')        // يكسر JSON strings
    .trim()
    .slice(0, 200);
}

// منع حقن في قيم JSON مباشرة
export function sanitizeForJson(str: string): string {
  return JSON.stringify(str).slice(1, -1); // escape كامل لـ JSON
}

// التطبيق: كل قيمة من المستخدم تمر عبر sanitizeForJson
```
